```typescript
import { ref, reactive } from 'vue'
import { Capacitor } from '@capacitor/core'
import { App as CapacitorApp } from '@capacitor/app'

const isNativePlatform = () => Capacitor.isNativePlatform()

/**
 * 与原生 GoProcessPlugin 通信的 Composable
 */
export function useGoProcess() {
  const isRestarting = ref(false)
  const isStopping = ref(false)
  const serviceStatus = reactive({
    running: false,
    port: 2025,
    lastError: '',
  })

  /**
   * 调用原生 GoProcessPlugin.restart() 方法
   * - iOS/Android 通过 Capacitor Plugin 架构调用
   * - Web 平台直接返回 true（无需操作）
   */
  async function restartService(): Promise<boolean> {
    if (!isNativePlatform()) {
      console.log('[useGoProcess] Web platform - restart skipped')
      return true
    }

    isRestarting.value = true
    try {
      // 动态导入 Plugin，避免 Web 平台加载错误
      const { GoProcess } = await import('@capacitor/plugins')

      console.log('[GoProcess] Calling restart()')
      const result = await GoProcess.restart()

      console.log('[GoProcess] Restart result:', result)
      serviceStatus.running = true
      serviceStatus.port = result.port || 2025
      serviceStatus.lastError = ''
      
      return true
    } catch (error) {
      const errMsg = error instanceof Error ? error.message : String(error)
      console.error('[GoProcess] Restart failed:', errMsg)
      serviceStatus.lastError = errMsg
      serviceStatus.running = false
      return false
    } finally {
      isRestarting.value = false
    }
  }

  /**
   * 调用原生 GoProcessPlugin.stop() 方法
   */
  async function stopService(): Promise<boolean> {
    if (!isNativePlatform()) {
      console.log('[useGoProcess] Web platform - stop skipped')
      return true
    }

    isStopping.value = true
    try {
      const { GoProcess } = await import('@capacitor/plugins')

      console.log('[GoProcess] Calling stop()')
      const result = await GoProcess.stop()

      console.log('[GoProcess] Stop result:', result)
      serviceStatus.running = false
      serviceStatus.lastError = ''
      
      return true
    } catch (error) {
      const errMsg = error instanceof Error ? error.message : String(error)
      console.error('[GoProcess] Stop failed:', errMsg)
      serviceStatus.lastError = errMsg
      return false
    } finally {
      isStopping.value = false
    }
  }

  /**
   * 获取后端服务当前状态
   */
  async function getStatus(): Promise<{ running: boolean; port: number }> {
    if (!isNativePlatform()) {
      return { running: true, port: 2025 }
    }

    try {
      const { GoProcess } = await import('@capacitor/plugins')
      const result = await GoProcess.getStatus()
      serviceStatus.running = result.running
      serviceStatus.port = result.port || 2025
      return result
    } catch (error) {
      console.error('[GoProcess] getStatus failed:', error)
      return { running: false, port: 0 }
    }
  }

  return {
    isRestarting,
    isStopping,
    serviceStatus,
    restartService,
    stopService,
    getStatus,
  }
}

export { isNativePlatform }
```

```typescript
import { ref, computed, onMounted, onUnmounted } from 'vue'
import { checkServerStatus } from '@/api/encv'
import { eventBus } from './useEventBus'
import { useGoProcess } from './useGoProcess'

let statusCheckInterval: number | null = null

export function useServerStatus() {
  const isOnline = ref(false)
  const lastError = ref('')
  const backendPort = ref(2025)
  const isChecking = ref(false)

  const { restartService, stopService, isRestarting, isStopping } = useGoProcess()

  /**
   * 检查服务器状态（包括 HTTP 请求 + 服务状态）
   */
  async function checkStatus() {
    if (isChecking.value) return

    isChecking.value = true
    try {
      const result = await checkServerStatus()
      isOnline.value = result.online
      lastError.value = result.error || ''

      if (result.online) {
        eventBus.emit('server:status', { online: true })
      } else {
        eventBus.emit('server:connection-error', { error: result.error || 'Connection failed' })
      }
    } catch (error) {
      isOnline.value = false
      const msg = error instanceof Error ? error.message : String(error)
      lastError.value = msg
      eventBus.emit('server:connection-error', { error: msg })
    } finally {
      isChecking.value = false
    }
  }

  /**
   * 重启后端服务
   * 1. 调用原生 Plugin 的 restart()
   * 2. 等待 2 秒后开始检查状态
   * 3. 轮询最多 10 秒直到连接成功
   */
  async function restartBackend(): Promise<boolean> {
    console.log('[useServerStatus] Restarting backend...')
    
    const success = await restartService()
    if (!success) {
      console.error('[useServerStatus] Service restart failed')
      return false
    }

    // 等待后端进程启动
    await new Promise(resolve => setTimeout(resolve, 2000))

    // 轮询检查连接
    for (let i = 0; i < 20; i++) {
      await checkStatus()
      if (isOnline.value) {
        console.log('[useServerStatus] Backend is now online')
        return true
      }
      await new Promise(resolve => setTimeout(resolve, 500))
    }

    console.warn('[useServerStatus] Backend did not come online within 10 seconds')
    return false
  }

  /**
   * 停止后端服务
   * 1. 调用原生 Plugin 的 stop()
   * 2. 等待 500ms 后检查状态
   */
  async function stopBackend(): Promise<boolean> {
    console.log('[useServerStatus] Stopping backend...')
    
    const success = await stopService()
    if (!success) {
      console.error('[useServerStatus] Service stop failed')
      return false
    }

    // 等待后端进程停止
    await new Promise(resolve => setTimeout(resolve, 500))
    await checkStatus()

    console.log('[useServerStatus] Backend stopped')
    return true
  }

  /**
   * 启动自动检查循环
   */
  function startAutoCheck() {
    if (statusCheckInterval) return

    // 立即检查一次
    checkStatus()

    // 每 5 秒检查一次
    statusCheckInterval = window.setInterval(() => {
      checkStatus()
    }, 5000)
  }

  /**
   * 停止自动检查
   */
  function stopAutoCheck() {
    if (statusCheckInterval) {
      clearInterval(statusCheckInterval)
      statusCheckInterval = null
    }
  }

  onMounted(() => {
    startAutoCheck()
  })

  onUnmounted(() => {
    stopAutoCheck()
  })

  return {
    isOnline,
    lastError,
    backendPort,
    isChecking,
    isRestarting,
    isStopping,
    checkStatus,
    restartBackend,
    stopBackend,
    startAutoCheck,
    stopAutoCheck,
  }
}
```

### 第三步：更新 `src/plugins/GoProcess.ts`（Capacitor Plugin 定义）

```typescript
import { registerPlugin } from '@capacitor/core'

export interface GoProcessPlugin {
  /**
   * 重启后端服务
   */
  restart(options?: any): Promise<{ success: boolean; port: number }>
  
  /**
   * 停止后端服务
   */
  stop(options?: any): Promise<{ success: boolean }>
  
  /**
   * 获取后端服务状态
   */
  getStatus(): Promise<{ running: boolean; port: number }>

  /**
   * 请求通知权限（Android 13+）
   */
  requestNotificationPermission(): Promise<{ granted: boolean }>

  /**
   * 请求存储权限（Android 11+）
   */
  requestStoragePermission(): Promise<{ granted: boolean; requiresSettings?: boolean }>

  /**
   * 检查权限状态
   */
  checkPermissions(): Promise<{ notifications: boolean; storage: boolean }>
}

const GoProcess = registerPlugin<GoProcessPlugin>('GoProcess')

export { GoProcess }

export function isNative(): boolean {
  return (window as any).Capacitor?.isNativePlatform() ?? false
}

export async function requestNotificationPermission() {
  try {
    if (!isNative()) return
    await GoProcess.requestNotificationPermission()
  } catch (e) {
    console.error('[GoProcess] requestNotificationPermission failed:', e)
  }
}

export async function requestStoragePermission() {
  try {
    if (!isNative()) return
    await GoProcess.requestStoragePermission()
  } catch (e) {
    console.error('[GoProcess] requestStoragePermission failed:', e)
  }
}

export async function checkPermissions() {
  try {
    if (!isNative()) {
      return { notifications: true, storage: true }
    }
    return await GoProcess.checkPermissions()
  } catch (e) {
    console.error('[GoProcess] checkPermissions failed:', e)
    return { notifications: false, storage: false }
  }
}
```

## 💡 核心特性

| 特性 | 实现 |
|------|------|
| **停止服务** | 调用 `GoProcess.stop()` → SERVICE.ACTION_STOP → destroyForcibly() |
| **重启服务** | 调用 `GoProcess.restart()` → SERVICE.ACTION_RESTART → 先 stop 再 start |
| **状态检查** | 定时轮询 HTTP `/health`，同时检查 Service 状态 |
| **UI 反馈** | 加载动画、Toast 提示、错误消息 |
| **权限管理** | Android 12+ FOREGROUND_SERVICE、13+ 通知权限 |
| **调试模式** | 双击标题切换调试信息（开发用） |

```kotlin
    const val ACTION_RESTART = "com.encvgo.action.RESTART"
}
```
