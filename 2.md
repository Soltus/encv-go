```xml
<?xml version="1.0" encoding="utf-8"?>
<manifest xmlns:android="http://schemas.android.com/apk/res/android">

    <!-- 基础权限（所有版本） -->
    <uses-permission android:name="android.permission.INTERNET" />
    
    <!-- ⭐ Android 12+ Foreground Service 权限 -->
    <uses-permission android:name="android.permission.FOREGROUND_SERVICE" />

    <!-- 对于数据处理/文件加密服务，用这个最宽松的 -->
    <uses-permission 
        android:name="android.permission.FOREGROUND_SERVICE_DATA_SYNC"
        android:minSdkVersion="33" />

    <!-- Android 14+ 允许精确定时唤醒（可选，如果需要定时检查） -->
    <uses-permission 
        android:name="android.permission.SCHEDULE_EXACT_ALARM"
        android:minSdkVersion="31" />

    <application>
        <!-- EncvGo 后台服务 -->
        <service
            android:name=".EncvGoService"
            android:foregroundServiceType="dataSync"
            android:enabled="true"
            android:exported="false" />

        <!-- MainActivity -->
        <activity
            android:name=".MainActivity"
            android:exported="true"
            android:launchMode="singleTop">
            <intent-filter>
                <action android:name="android.intent.action.MAIN" />
                <category android:name="android.intent.category.LAUNCHER" />

        </activity>
    </application>

```

**关键点：**
- ✅ `android:foregroundServiceType="dataSync"` - 最宽松的类型，适合数据处理
- ✅ `android:minSdkVersion` 属性避免低版本崩溃
- ✅ `android:exported="false"` - 仅 APP 内部使用

### 第二步：EncvGoService.kt（后台服务）

```kotlin
package com.encvgo.app

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Context
import android.content.Intent
import android.os.Build
import android.os.IBinder
import android.util.Log
import androidx.core.app.NotificationCompat
import java.io.BufferedReader
import java.io.File
import java.io.FileOutputStream
import java.io.InputStreamReader

class EncvGoService : Service() {
    companion object {
        private const val TAG = "ENCV-Service"
        private const val NOTIFICATION_ID = 1001
        private const val CHANNEL_ID = "encv_go_channel"
        private const val BINARY_NAME = "encv-go"
        private const val ACTION_START = "com.encvgo.action.START"
        private const val ACTION_STOP = "com.encvgo.action.STOP"
        private const val ACTION_RESTART = "com.encvgo.action.RESTART"
    }

    private var goProcess: Process? = null
    private var isRunning = false
    private var notificationManager: NotificationManager? = null

    override fun onCreate() {
        super.onCreate()
        Log.d(TAG, "Service created")
        notificationManager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        createNotificationChannel()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        Log.d(TAG, "onStartCommand: action=${intent?.action}")
        
        return when (intent?.action) {
            ACTION_START -> {
                startEncvGo()
                START_STICKY  // 系统杀掉后自动重启
            }
            ACTION_RESTART -> {
                stopEncvGo()
                Thread.sleep(500)
                startEncvGo()
                START_STICKY
            }
            ACTION_STOP -> {
                stopEncvGo()
                stopSelf()
                START_NOT_STICKY
            }
            else -> {
                if (!isRunning) {
                    startEncvGo()
                }
                START_STICKY
            }
        }
    }

    private fun startEncvGo() {
        if (isRunning && goProcess?.isAlive == true) {
            Log.w(TAG, "Already running")
            return
        }

        try {
            // 提前启动前台通知（Android 12+ 要求在启动 Service 后 5 秒内调用 startForeground）
            val notification = buildNotification("正在启动后端...")
            startForeground(NOTIFICATION_ID, notification)

            val binary = extractBinary()
            if (binary == null) {
                Log.e(TAG, "Failed to extract binary")
                updateNotification("启动失败：二进制文件缺失")
                return
            }

            val configPath = File(filesDir, "config.user.json").absolutePath
            ensureConfigExists()

            val cmd = "${binary.absolutePath} start"
            Log.i(TAG, "Starting: $cmd")
            
            goProcess = ProcessBuilder("/system/bin/sh", "-c", cmd).apply {
                environment()["ENCV_CONFIG_PATH"] = configPath
                environment()["ENCV_MOBILE"] = "1"
                environment()["HOME"] = filesDir.absolutePath
                redirectErrorStream(true)
                directory(filesDir)
            }.start()

            isRunning = true
            updateNotification("后端已启动")

            // 监听进程输出
            Thread {
                monitorProcess()
            }.start()

        } catch (e: Exception) {
            Log.e(TAG, "Failed to start Go daemon", e)
            updateNotification("启动失败：${e.message?.take(50)}")
        }
    }

    private fun stopEncvGo() {
        goProcess?.let {
            if (it.isAlive) {
                it.destroyForcibly()
                Log.i(TAG, "Go daemon stopped")
            }
        }
        isRunning = false
        updateNotification("已停止")
    }

    private fun monitorProcess() {
        try {
            val reader = BufferedReader(InputStreamReader(goProcess?.inputStream))
            var line: String?
            while (reader.readLine().also { line = it } != null) {
                Log.i(TAG, "[go] $line")
                
                // 关键状态检测
                if (line?.contains("listening on", ignoreCase = true) == true ||
                    line?.contains("server ready", ignoreCase = true) == true) {
                    updateNotification("后端已就绪 ✓")
                    // 发送 Broadcast 通知 MainActivity
                    sendBroadcast(Intent("com.encvgo.BACKEND_READY").apply {
                        setPackage(packageName)
                    })
                }
                
                if (line?.contains("error", ignoreCase = true) == true ||
                    line?.contains("fatal", ignoreCase = true) == true) {
                    Log.e(TAG, "Backend error: $line")
                    updateNotification("后端错误：${line?.take(50)}")
                }
            }
            
            val exitCode = goProcess?.waitFor() ?: -1
            Log.w(TAG, "Process exited with code: $exitCode")
            isRunning = false
            updateNotification("后端已退出 (code:$exitCode)")
            
        } catch (e: Exception) {
            Log.w(TAG, "Error monitoring process", e)
        }
    }

    private fun extractBinary(): File? {
        val binary = File(filesDir, BINARY_NAME)
        
        if (!binary.exists()) {
            try {
                assets.open(BINARY_NAME).use { input ->
                    FileOutputStream(binary).use { output ->
                        input.copyTo(output)
                    }
                }
                Log.i(TAG, "Extracted binary: ${binary.absolutePath}")
            } catch (e: Exception) {
                Log.e(TAG, "Failed to extract binary", e)
                return null
            }
        }
        
        binary.setExecutable(true)
        binary.setReadable(true)
        return if (binary.canExecute()) binary else null
    }

    private fun ensureConfigExists() {
        val dest = File(filesDir, "config.user.json")
        if (!dest.exists()) {
            try {
                assets.open("config.mobile.json").use { input ->
                    FileOutputStream(dest).use { output ->
                        input.copyTo(output)
                    }
                }
                Log.i(TAG, "Copied default config")
            } catch (e: Exception) {
                Log.e(TAG, "Failed to copy config", e)
            }
        }
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "ENCV-go 后端服务",
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = "ENCV-go 数据加密服务"
                setShowBadge(false)
            }
            notificationManager?.createNotificationChannel(channel)
        }
    }

    private fun buildNotification(text: String): Notification {
        val intent = Intent(this, MainActivity::class.java)
        val pendingIntent = PendingIntent.getActivity(
            this, 0, intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("ENCV-go")
            .setContentText(text)
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setContentIntent(pendingIntent)
            .setOngoing(true)
            .build()
    }

    private fun updateNotification(text: String) {
        notificationManager?.notify(NOTIFICATION_ID, buildNotification(text))
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        super.onDestroy()
        stopEncvGo()
        Log.d(TAG, "Service destroyed")
    }
}
```

### 第三步：改进版 MainActivity.kt

```kotlin
package com.encvgo.app

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.os.Build
import android.os.Bundle
import android.util.Log
import androidx.localbroadcastmanager.content.LocalBroadcastManager
import com.getcapacitor.BridgeActivity

class MainActivity : BridgeActivity() {
    companion object {
        private const val TAG = "ENCV-MainActivity"
    }

    private var backendReady = false
    private var backendPort = 2025
    private val backendReceiver = object : BroadcastReceiver() {
        override fun onReceive(context: Context?, intent: Intent?) {
            Log.d(TAG, "Backend ready broadcast received")
            backendReady = true
            notifyFrontendReady(backendPort)
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        Log.d(TAG, "=== onCreate ===")
        try {
            registerPlugin(GoProcessPlugin::class.java)
            Log.d(TAG, "GoProcessPlugin registered")
        } catch (e: Exception) {
            Log.e(TAG, "registerPlugin failed", e)
        }

        super.onCreate(savedInstanceState)
        
        // 注册 Broadcast 接收器
        val filter = IntentFilter("com.encvgo.BACKEND_READY")
        LocalBroadcastManager.getInstance(this).registerReceiver(backendReceiver, filter)

        // 启动后台服务
        startBackendService()
    }

    private fun startBackendService() {
        val serviceIntent = Intent(this, EncvGoService::class.java).apply {
            action = EncvGoService.ACTION_START  // 需要在 companion object 中添加
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            startForegroundService(serviceIntent)  // Android 8+ 必须用 startForegroundService
        } else {
            startService(serviceIntent)
        }

        Log.d(TAG, "Backend service started")
        
        // 稍微等一下再通知前端（给服务一点时间启动）
        Thread {
            Thread.sleep(2000)
            if (backendReady) {
                notifyFrontendReady(backendPort)
            }
        }.start()
    }

    private fun notifyFrontendReady(port: Int) {
        runOnUiThread {
            try {
                val js = "window.dispatchEvent(new CustomEvent('encv:backend-ready',{detail:{port:$port}}))"
                bridge.webView.evaluateJavascript(js, null)
                Log.d(TAG, "Frontend notified: port=$port")
            } catch (e: Exception) {
                Log.w(TAG, "Failed to notify frontend", e)
            }
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        LocalBroadcastManager.getInstance(this).unregisterReceiver(backendReceiver)
        // ⚠️ 不要在 onDestroy 中停止 Service，让它继续在后台运行
        Log.d(TAG, "Activity destroyed (service still running)")
    }
}
```

```kotlin
package com.encvgo.app

import android.content.Intent
import android.util.Log
import com.getcapacitor.JSObject
import com.getcapacitor.Plugin
import com.getcapacitor.PluginCall
import com.getcapacitor.PluginMethod
import com.getcapacitor.annotation.CapacitorPlugin

@CapacitorPlugin(name = "GoProcess")
class GoProcessPlugin : Plugin() {

    companion object {
        private const val TAG = "ENCV-Plugin"
    }

    @PluginMethod
    fun getStatus(call: PluginCall) {
        Log.d(TAG, "getStatus called")
        // TODO: 从 Service 获取实时状态
        // 这里可以通过 LocalBroadcastManager 或 ViewModel 获取
        val result = JSObject()
        result.put("running", true)
        result.put("port", 2025)
        call.resolve(result)
    }

    @PluginMethod
    fun restart(call: PluginCall) {
        Log.d(TAG, "restart called")
        val serviceIntent = Intent(context, EncvGoService::class.java).apply {
            // 在 EncvGoService 中添加 ACTION_RESTART 常量
            action = "com.encvgo.action.RESTART"
        }
        context.startService(serviceIntent)
        call.resolve(JSObject().apply { put("success", true) })
    }

    @PluginMethod
    fun stop(call: PluginCall) {
        Log.d(TAG, "stop called")
        val serviceIntent = Intent(context, EncvGoService::class.java).apply {
            action = "com.encvgo.action.STOP"
        }
        context.startService(serviceIntent)
        call.resolve(JSObject().apply { put("success", true) })
    }
}
```

## ⚠️ 注意事项

1. **权限检查时机**：
   ```kotlin
   // Android 13+ 需要在运行时检查
   if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
       // 检查具体的子权限
   }
   ```

2. **通知持久化**：
   - Foreground Service 需要始终显示通知
   - 用户可以滑动关闭，但 Service 继续运行

3. **省电模式**：
   - Android 12+ Doze mode 中，Foreground Service 优先级较高
   - 但仍可能被系统限制，需要 `SCHEDULE_EXACT_ALARM` 配合

4. **Google Play 审核**：
   - 申请 `SCHEDULE_EXACT_ALARM` 需要有合理理由
   - `FOREGROUND_SERVICE_DATA_SYNC` 是最宽松的，通常能过审
